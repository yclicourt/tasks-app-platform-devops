backend/
backend-deployment.yml
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: backend-app-ingress
  namespace: {{.Values.backend_tasks_app.namespace}}
  labels:
    app.kubernetes.io/name: backend-app-ingress
  annotations:
    kubernetes.io/ingress.class: traefik
    traefik.ingress.kubernetes.io/router.entrypoints: web
    # Ayuda con problemas de timeout y buffers
    traefik.ingress.kubernetes.io/service.serversscheme: http
    traefik.ingress.kubernetes.io/router.priority: "10"
spec:
  rules:
    - host: api-tasks.local
      http:
        paths:
          - path: /
            pathType: Prefix
            backend:
              service:
                name: {{.Values.backend_tasks_app.serviceName}}
                port:
                  number: 8000
---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: {{.Values.backend_tasks_app.name}}
  namespace: {{.Values.backend_tasks_app.namespace}}
spec:
  replicas: 1
  selector:
    matchLabels:
      app: {{.Values.backend_tasks_app.name}}
  template:
    metadata:
      labels:
        app: {{.Values.backend_tasks_app.name}}
    spec:
      serviceAccountName: {{.Values.backend_tasks_app.serviceAccountName}}
      initContainers:
        - name: run-migrations
          image: "{{ .Values.backend_tasks_app.image.repository}}:{{.Values.backend_tasks_app.image.tag}}"
          command: ['python', 'manage.py', 'migrate', '--noinput']
          envFrom:
            - secretRef:
                name: {{.Values.backend_tasks_app.secretName}}
      containers:
        - name: {{.Values.backend_tasks_app.name}}
          image: "{{ .Values.backend_tasks_app.image.repository}}:{{.Values.backend_tasks_app.image.tag}}"
          imagePullPolicy: Always
          envFrom:
            - secretRef:
                name: {{.Values.backend_tasks_app.secretName}}
          resources:
            requests:
              memory: "256Mi"
              cpu: "500m"
            limits:
              memory: "512Mi"
              cpu: "1000m"
          ports:
            - containerPort: 8000
---
apiVersion: v1
kind: Service
metadata:
  name: {{.Values.backend_tasks_app.serviceName}}
  namespace: {{.Values.backend_tasks_app.namespace}}
spec:
  type: ClusterIP
  selector:
    app: {{.Values.backend_tasks_app.name}}
  ports:
    - port: 8000
      targetPort: 8000

postgres-statefulset.yml

apiVersion: apps/v1
kind: StatefulSet
metadata:
  name: {{.Values.tasks_db.name}}
  namespace: {{.Values.tasks_db.namespace}}
spec:
  selector:
    matchLabels:
      app: {{.Values.tasks_db.name}}
  serviceName: {{.Values.tasks_db.serviceName}}
  replicas: 1
  template:
    metadata:
      labels:
        app: {{.Values.tasks_db.name}}
    spec:
      securityContext:
        fsGroup: 70
      serviceAccountName: {{.Values.tasks_db.serviceAccountName}}
      containers:
        - name: {{.Values.tasks_db.name}}
          image: "{{ .Values.tasks_db.image.repository }}:{{ .Values.tasks_db.image.tag }}"
          env:
            - name: POSTGRES_DB
              valueFrom:
                secretKeyRef:
                  name: {{.Values.tasks_db.secretName}}
                  key: POSTGRES_DB
            - name: POSTGRES_USER
              valueFrom:
                secretKeyRef:
                  name: {{.Values.tasks_db.secretName}}
                  key: POSTGRES_USER
            - name: POSTGRES_PASSWORD
              valueFrom:
                secretKeyRef:
                  name: {{.Values.tasks_db.secretName}}
                  key: POSTGRES_PASSWORD
          ports:
            - containerPort: 5432
          volumeMounts:
            - name: {{.Values.tasks_db.volumeName}}
              mountPath: /var/lib/postgresql/data
  volumeClaimTemplates:
    - metadata:
        name: {{.Values.tasks_db.volumeName}}
      spec:
        accessModes: ["ReadWriteOnce"]
        storageClassName: local-path
        resources:
          requests:
            storage: 2Gi

---
apiVersion: v1
kind: Service
metadata:
  name: {{.Values.tasks_db.serviceName}}
  namespace: {{.Values.tasks_db.namespace}}
spec:
  clusterIP: None
  selector:
    app: {{.Values.tasks_db.name}}
  ports:
    - port: 5432
      targetPort: 5432
